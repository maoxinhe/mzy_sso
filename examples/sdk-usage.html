登录方式：

// 核心代码只有 4 行
const sso = MZYSSO.init({
  issuer:   'https://sso.camzy.uno',
  clientId: 'mzy_xxxxxxxxxxxx'
});

sso.login();                            // 跳转登录
const s = await sso.handleRedirect();   // 回调页处理
const { user } = await sso.getUser();   // 获取用户（自动续期）
sso.logout();                           // 退出
